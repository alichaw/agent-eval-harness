# Human-operated T3 Windows SSH acceptance runbook

Status: generated by Codex and not executed. This research PoC has exactly two
live-capable actions and no marker action:

- `t3a.ssh22_reachability.v1`
- `windows.ssh.readonly_identity.v1`

The first Windows connection occurs only in Case 3. Stop after any failed case.
Never return protected values or raw logs. The sole protected migration input is
`config/local/t3-runtime.json`; the prepare tool derives, validates, and atomically
writes `/etc/hexstrike/t3-unified-runtime.json` without accepting protected values
in `argv` or overwriting the legacy input.

# LIVE ACTION — HUMAN AUTHORIZATION REQUIRED

## Prerequisite Setup — Human Operator Only

This setup performs no target connection, SSH authentication, approval operation,
or evidence creation. The `PROVISION` prompt is the only interactive confirmation.
It loads the protected key through stdin into the dedicated agent and never prints
the key, its path, fingerprint, or protected runtime values.

```bash
# LIVE ACTION — HUMAN OPERATOR ONLY — DO NOT EXECUTE BY CODEX
set -eu
HARNESS_ROOT=/home/kali/agent-eval-harness
HEXSTRIKE_ROOT=/home/kali/hexstrike-ai
LEGACY_RUNTIME="$HARNESS_ROOT/config/local/t3-runtime.json"
UNIFIED_RUNTIME=/etc/hexstrike/t3-unified-runtime.json

test ! -L "$LEGACY_RUNTIME"
test "$(stat -c %a "$LEGACY_RUNTIME")" = 600
test "$(stat -c %u "$LEGACY_RUNTIME")" = "$(id -u)"

sudo "$HARNESS_ROOT/scripts/install_hexstrike_t3_poc.sh" --bootstrap-identity
sudo "$HARNESS_ROOT/.venv/bin/python" "$HARNESS_ROOT/scripts/prepare_t3_poc_protected_config.py"
sudo env HEXSTRIKE_SOURCE_DIR="$HEXSTRIKE_ROOT" "$HARNESS_ROOT/scripts/install_hexstrike_t3_poc.sh"
sudo systemctl restart hexstrike-t3-poc.service
sudo systemctl is-active --quiet hexstrike-t3-poc.service
sudo systemctl is-active --quiet hexstrike-t3-ssh-agent.service

# INTERACTIVE HUMAN CONFIRMATION: type PROVISION when prompted.
sudo "$HARNESS_ROOT/.venv/bin/python" "$HARNESS_ROOT/scripts/provision_t3_poc_ssh_agent.py" \
  --legacy-runtime-file "$LEGACY_RUNTIME"

test ! -L "$UNIFIED_RUNTIME"
test "$(stat -c %U:%G:%a "$UNIFIED_RUNTIME")" = hexstrike:hexstrike:600
test "$(stat -c %U:%G:%a /var/lib/hexstrike/t3-unified-approvals-v2.sqlite3)" = hexstrike:hexstrike:600
test "$(stat -c %U:%G:%a /var/lib/hexstrike/t3-unified-evidence)" = hexstrike:hexstrike:700
test "$(stat -c %U:%G:%a /run/hexstrike)" = hexstrike:hexstrike:750
test -S /run/hexstrike-t3-ssh-agent/agent.sock
test ! -L /run/hexstrike-t3-ssh-agent/agent.sock
sudo -u hexstrike env SSH_AUTH_SOCK=/run/hexstrike-t3-ssh-agent/agent.sock ssh-add -l \
  >/dev/null 2>&1

sudo env HARNESS_ROOT="$HARNESS_ROOT" HEXSTRIKE_ROOT="$HEXSTRIKE_ROOT" \
  LEGACY_RUNTIME="$LEGACY_RUNTIME" UNIFIED_RUNTIME="$UNIFIED_RUNTIME" \
  "$HARNESS_ROOT/.venv/bin/python" - <<'PY'
import hashlib, json, os, pwd, stat, sys
from pathlib import Path
sys.path.insert(0, os.environ["HARNESS_ROOT"])
from core.t3.poc_registry import ACTIONS
from core.t3.poc_runtime import load_runtime
account = pwd.getpwnam("hexstrike")
runtime = load_runtime(os.environ["UNIFIED_RUNTIME"], owner_uid=account.pw_uid)
assert list(ACTIONS) == ["t3a.ssh22_reachability.v1", "windows.ssh.readonly_identity.v1"]
assert runtime.action_policy["enabled_action_ids"] == tuple(ACTIONS)
legacy = json.loads(Path(os.environ["LEGACY_RUNTIME"]).read_text())
key_path = Path(legacy["private_key_path"]).resolve()
assert key_path.is_file() and not key_path.is_symlink()
assert not any(root == key_path or root in key_path.parents for root in (
    Path(os.environ["HARNESS_ROOT"]).resolve(), Path(os.environ["HEXSTRIKE_ROOT"]).resolve(),
    Path("/opt/hexstrike-t3-poc").resolve(), Path("/etc/hexstrike").resolve()))
runtime_bytes = Path(os.environ["UNIFIED_RUNTIME"]).read_bytes()
key_bytes = key_path.read_bytes()
assert b"private_key_path" not in runtime_bytes and key_bytes not in runtime_bytes
key_digest = hashlib.sha256(key_bytes).digest()
for root in (Path(os.environ["HARNESS_ROOT"]), Path(os.environ["HEXSTRIKE_ROOT"]),
             Path("/opt/hexstrike-t3-poc"), Path("/etc/hexstrike")):
    for candidate in root.rglob("*"):
        try:
            if candidate.is_file() and candidate.stat().st_size == len(key_bytes):
                assert hashlib.sha256(candidate.read_bytes()).digest() != key_digest
        except PermissionError:
            raise AssertionError("protected_copy_verification_failed") from None
assert stat.S_IMODE(Path(runtime.approval_database).stat().st_mode) == 0o600
assert stat.S_IMODE(Path(runtime.evidence_root).stat().st_mode) == 0o700
assert not Path(runtime.kill_switch_file).exists()
print("Prerequisite Setup PASS: minimal service, protected state, two-action runtime, and agent identity verified")
PY
```

Stop immediately on any failure. Do not proceed to Case 0 if provisioning,
ownership, mode, symlink, service identity, agent verification, or configuration
validation fails.

## Case 3 — Windows TCP/22 reachability

Contacts Windows: yes—this is the first contact. Creates approval: yes. Consumes
approval: yes when the execution boundary begins. Evidence:
`<evidence_root>/<authorization_digest>/<action_digest>.json`.

```bash

HARNESS_ROOT=/home/kali/agent-eval-harness
RUNTIME_FILE=/etc/hexstrike/t3-unified-runtime.json

sudo -u hexstrike env PYTHONPATH="$HARNESS_ROOT" "$HARNESS_ROOT/.venv/bin/python" -m core.cli \
  t3-unified-authorize-reachability --runtime-file "$RUNTIME_FILE" --authorization-file "$AUTH_FILE"
AUTH_CREATED=1
sudo -u hexstrike env PYTHONPATH="$HARNESS_ROOT" "$HARNESS_ROOT/.venv/bin/python" -m core.cli \
  t3-unified-run --runtime-file "$RUNTIME_FILE" --authorization-file "$AUTH_FILE"
sudo -u hexstrike env RUNTIME_FILE="$RUNTIME_FILE" AUTH_FILE="$AUTH_FILE" ACTION_ID=t3a.ssh22_reachability.v1 \
  "$HARNESS_ROOT/.venv/bin/python" - <<'PY'
import hashlib, json, os, sqlite3, sys
sys.path.insert(0, "/home/kali/agent-eval-harness")
from core.t3.poc_runner import load_authorization
from core.t3.poc_runtime import load_runtime
```

## Case 5 — Kill-switch denial and audit verification

Contacts Windows after activation: no. Creates approval: yes, before activation.
Consumes approval: no. Evidence: `<evidence_root>/audit/<event_id>.json`; approval
must remain pending. Cleanup removes the kill-switch and authorization file.

```bash
# LIVE ACTION — HUMAN OPERATOR ONLY — DO NOT EXECUTE BY CODEX
set -eu
HARNESS_ROOT=/home/kali/agent-eval-harness
RUNTIME_FILE=/etc/hexstrike/t3-unified-runtime.json
AUTH_FILE=/run/hexstrike/t3-kill-denial-authorization.json
KILL_FILE=/run/hexstrike/KILL
test ! -e "$AUTH_FILE"
test ! -e "$KILL_FILE"
AUTH_CREATED=0
KILL_CREATED=0
cleanup() {
  test "$KILL_CREATED" -eq 0 || sudo rm -f -- "$KILL_FILE"
  test "$AUTH_CREATED" -eq 0 || sudo -u hexstrike rm -f -- "$AUTH_FILE"
}
trap cleanup EXIT
sudo -u hexstrike env PYTHONPATH="$HARNESS_ROOT" "$HARNESS_ROOT/.venv/bin/python" -m core.cli \
  t3-unified-authorize-identity --runtime-file "$RUNTIME_FILE" --authorization-file "$AUTH_FILE"
AUTH_CREATED=1
sudo install -o hexstrike -g hexstrike -m 0600 /dev/null "$KILL_FILE"
KILL_CREATED=1
set +e
DENIAL_OUTPUT=$(sudo -u hexstrike env PYTHONPATH="$HARNESS_ROOT" "$HARNESS_ROOT/.venv/bin/python" -m core.cli \
  t3-unified-run --runtime-file "$RUNTIME_FILE" --authorization-file "$AUTH_FILE" 2>&1)
DENIAL_STATUS=$?
set -e
test "$DENIAL_STATUS" -ne 0
test "$DENIAL_OUTPUT" = kill_switch_engaged
sudo -u hexstrike env RUNTIME_FILE="$RUNTIME_FILE" AUTH_FILE="$AUTH_FILE" \
  "$HARNESS_ROOT/.venv/bin/python" - <<'PY'
import hashlib, json, os, sqlite3, sys
sys.path.insert(0, "/home/kali/agent-eval-harness")
from core.t3.poc_runner import load_authorization
from core.t3.poc_runtime import load_runtime
runtime = load_runtime(os.environ["RUNTIME_FILE"])
raw = load_authorization(os.environ["AUTH_FILE"])
digest = hashlib.sha256(raw.encode()).hexdigest()
with sqlite3.connect(runtime.approval_database) as db:
    state = db.execute("SELECT state FROM approvals WHERE authorization_id_digest=?", (digest,)).fetchone()
assert state == ("pending",)
matches = []
for path in (__import__("pathlib").Path(runtime.evidence_root) / "audit").glob("*.json"):
    candidate = json.loads(path.read_text())
    if (candidate.get("authorization_id_digest") == digest
            and candidate.get("action_id") == "windows.ssh.readonly_identity.v1"
            and candidate.get("runtime_digest") == runtime.digest
            and candidate.get("policy_decision") == "kill_switch_denied"):
        matches.append(candidate)
assert len(matches) == 1
event = matches[0]; seal = event.pop("seal")
canonical = json.dumps(event, sort_keys=True, separators=(",", ":"), ensure_ascii=True)
assert seal == hashlib.sha256(canonical.encode()).hexdigest()
print("Case 5 audit PASS: kill-switch denial sealed; approval remains pending")
PY
echo 'Case 5 PASS: kill-switch blocked connection and denial audit was verified'
```

Stop immediately if the denial is not `kill_switch_engaged`, approval is consumed,
no sealed denial event exists, any Windows connection occurs after activation, or
cleanup fails. Do not proceed or retry while the switch exists.
